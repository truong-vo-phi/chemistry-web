// DATABASE VERSION 2 - Tối ưu cho Web thực tế

Table Users {
  id integer [primary key]
  username varchar [unique]
  password varchar
  full_name varchar
  email varchar
  avatar_url text // Thêm ảnh đại diện
  role varchar // 'HS', 'Teacher', 'Admin'
  school_id integer 
  is_active boolean [default: true]
  created_at timestamp
}

Table Schools {
  id integer [primary key]
  name varchar
  address text
}

Table Courses {
  id integer [primary key]
  title varchar
  slug varchar [unique] // Dùng cho URL web (ví dụ: /khoa-hoc/toan-lop-10)
  description text
  thumbnail_url text
  teacher_id integer [ref: > Users.id]
  status varchar [default: 'draft'] // 'draft', 'published', 'archived'
  created_at timestamp
}

// Thêm Chương để quản lý bài học tốt hơn
Table Chapters {
  id integer [primary key]
  course_id integer [ref: > Courses.id]
  title varchar
  order_index integer // Thứ tự chương
}

Table Lessons {
  id integer [primary key]
  chapter_id integer [ref: > Chapters.id]
  title varchar
  content_type varchar // 'Video', 'Document', 'Practice'
  video_url text
  document_content text
  order_index integer // Thứ tự bài học
  created_at timestamp
}

// QUAN TRỌNG: Theo dõi học sinh đã hoàn thành bài nào
Table UserLessonProgress {
  user_id integer [ref: > Users.id]
  lesson_id integer [ref: > Lessons.id]
  is_completed boolean [default: false]
  completed_at timestamp
  primary key (user_id, lesson_id)
}

Table CourseEnrollments {
  course_id integer [ref: > Courses.id]
  student_id integer [ref: > Users.id]
  enrolled_at timestamp
  primary key (course_id, student_id)
}

Table Reviews {
  id integer [primary key]
  course_id integer [ref: > Courses.id]
  user_id integer [ref: > Users.id]
  content text
  rating integer
  created_at timestamp
}

Table Revisions {
  id integer [primary key]
  course_id integer [ref: > Courses.id]
  title varchar
  time_limit integer // Thời gian làm bài (phút)
  created_at timestamp
}

Table Questions {
  id integer [primary key]
  revision_id integer [ref: > Revisions.id]
  content text
  options json // Lưu các lựa chọn A, B, C, D dưới dạng JSON
  correct_answer text
}

Table QuizResults {
  id integer [primary key]
  revision_id integer [ref: > Revisions.id]
  student_id integer [ref: > Users.id]
  score float
  total_correct integer
  completed_at timestamp
}

Table News {
  id integer [primary key]
  title varchar
  slug varchar
  content text
  thumbnail_url text
  author_id integer [ref: > Users.id]
  status varchar [default: 'published']
  created_at timestamp
}

Table Classes {
  id integer [primary key]
  class_name varchar
  school_id integer [ref: > Schools.id]
  teacher_id integer [ref: > Users.id]
  created_at timestamp
}

Table ClassMembers {
  class_id integer [ref: > Classes.id]
  student_id integer [ref: > Users.id]
  primary key (class_id, student_id)
}

Ref: Users.school_id > Schools.id
